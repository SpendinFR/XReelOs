/**
 * Minimal Shizuku UserService used only by the isolated secure-surface spike.
 * It runs as uid=shell, creates a trusted VirtualDisplay backed directly by the
 * XREAL protected Surface, and launches packaged Android applications there.
 */

    @SuppressWarnings("unused")
    public TrustedDisplayUserService(Context context) {
        suppliedContext = context;
        attachInterface(null, DESCRIPTOR);
        setupInputInjection();
        Log.i(TAG, "service created uid=" + Process.myUid());
    }

    @Override
    protected boolean onTransact(int code, Parcel data, Parcel reply, int flags) {
        try {
            if (code == INTERFACE_TRANSACTION) {
                reply.writeString(DESCRIPTOR);
                return true;
            }
            data.enforceInterface(DESCRIPTOR);
            if (code == TRANSACTION_CREATE) {
                String name = data.readString();
                int width = data.readInt();
                int height = data.readInt();
                int dpi = data.readInt();
                int displayFlags = data.readInt();
                Surface surface = Surface.CREATOR.createFromParcel(data);
                int displayId = createDisplay(
                        name, width, height, dpi, displayFlags, surface);
                reply.writeNoException();
                reply.writeInt(displayId);
                return true;
            }
            if (code == TRANSACTION_LAUNCH_YOUTUBE) {
                int displayId = data.readInt();
                String uri = data.readString();
                String result = launchYoutube(displayId, uri);
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_RELEASE) {
                releaseDisplay();
                reply.writeNoException();
                terminateAfterReply();
                return true;
            }
            if (code == TRANSACTION_POINTER) {
                int displayId = data.readInt();
                int action = data.readInt();
                float x = data.readFloat();
                float y = data.readFloat();
                long downTime = data.readLong();
                boolean injected = injectPointer(
                        displayId, action, x, y, downTime);
                reply.writeNoException();
                reply.writeInt(injected ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_HOVER) {
                int displayId = data.readInt();
                float x = data.readFloat();
                float y = data.readFloat();
                boolean injected = injectHover(displayId, x, y);
                reply.writeNoException();
                reply.writeInt(injected ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_OPEN_KEYBOARD) {
                int displayId = data.readInt();
                boolean opened = openKeyboard(displayId);
                reply.writeNoException();
                reply.writeInt(opened ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_LAUNCH_APP) {
                int displayId = data.readInt();
                String packageName = data.readString();
                String uri = data.readString();
                String result = launchPackagedUri(displayId, packageName, uri);
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_START_SECURE_STEREO_TASK) {
                Surface spatialSurface = Surface.CREATOR.createFromParcel(data);
                int surfaceWidth = data.readInt();
                int surfaceHeight = data.readInt();
                String packageName = data.readString();
                String uri = data.readString();
                String result = startSecureStereoTask(
                        spatialSurface,
                        surfaceWidth,
                        surfaceHeight,
                        packageName,
                        uri);
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_MAYBE_START_SYSTEM_MIRROR) {
                int glassesDisplayId = data.readInt();
                int virtualDisplayId = data.readInt();
                String packageName = data.readString();
                boolean force = data.readInt() != 0;
                String result = maybeStartSystemMirror(
                        glassesDisplayId,
                        virtualDisplayId,
                        packageName,
                        force);
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_STOP_SYSTEM_MIRROR) {
                String result = stopSystemMirror();
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_REATTACH_SURFACE) {
                Surface surface = Surface.CREATOR.createFromParcel(data);
                boolean attached = reattachDisplaySurface(surface);
                reply.writeNoException();
                reply.writeInt(attached ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_SET_CINEMA_PROJECTION) {
                int preset = data.readInt();
                String result = applyCinemaProjection(preset);
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_IS_PLAYBACK_READY) {
                String packageName = data.readString();
                boolean ready = isProtectedPlaybackActive(packageName);
                reply.writeNoException();
                reply.writeInt(ready ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_TOGGLE_MEDIA) {
                String result = toggleMirroredMedia();
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_STOP_PACKAGE) {
                String packageName = data.readString();
                boolean stopped = forceStopPackage(packageName);
                reply.writeNoException();
                reply.writeInt(stopped ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_INPUT_TEXT) {
                int displayId = data.readInt();
                String text = data.readString();
                boolean inserted = inputText(displayId, text);
                reply.writeNoException();
                reply.writeInt(inserted ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_INPUT_KEY) {
                int displayId = data.readInt();
                int keyCode = data.readInt();
                boolean inserted = inputKey(displayId, keyCode);
                reply.writeNoException();
                reply.writeInt(inserted ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_CLEAR_TEXT) {
                int displayId = data.readInt();
                boolean cleared = clearText(displayId);
                reply.writeNoException();
                reply.writeInt(cleared ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_RESIZE_DISPLAY) {
                int width = data.readInt();
                int height = data.readInt();
                int dpi = data.readInt();
                boolean resized = resizeDisplay(width, height, dpi);
                reply.writeNoException();
                reply.writeInt(resized ? 1 : 0);
                return true;
            }
            if (code == TRANSACTION_SEEK_MEDIA_BACK) {
                long milliseconds = data.readLong();
                String result = seekMirroredMediaBack(milliseconds);
                reply.writeNoException();
                reply.writeString(result);
                return true;
            }
            if (code == TRANSACTION_RUNTIME_PREFLIGHT) {
                String result = performRuntimePreflight();
                reply.writeNoException();
                reply.writeString(result);
                terminateAfterReply();
                return true;
            }
            return super.onTransact(code, data, reply, flags);
        } catch (Throwable error) {
            Log.e(TAG, "transaction " + code + " failed", error);
            reply.writeException(new IllegalStateException(
                    error.getClass().getSimpleName() + ": " + error.getMessage()));
            return true;
        }
    }

    private int createDisplay(
            String name,
            int width,
            int height,
            int dpi,
            int flags,
            Surface surface) throws Exception {
        releaseDisplay();
        long identity = Binder.clearCallingIdentity();
        try {
            Context context = obtainShellContext();
            if (context == null) {
                throw new IllegalStateException("shell context unavailable");
            }
            DisplayManager manager = (DisplayManager) context.getSystemService(
                    Context.DISPLAY_SERVICE);
            if (manager == null) {
                throw new IllegalStateException("DisplayManager unavailable");
            }
            virtualDisplay = manager.createVirtualDisplay(
                    name,
                    width,
                    height,
                    dpi,
                    surface,
                    flags);
            if (virtualDisplay == null || virtualDisplay.getDisplay() == null) {
                throw new IllegalStateException("trusted VirtualDisplay creation failed");
            }
            displaySurface = surface;
            int id = virtualDisplay.getDisplay().getDisplayId();
            Log.i(TAG, "created display=" + id + " flags=0x"
                    + Integer.toHexString(flags) + " uid=" + Process.myUid());
            return id;
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    /**
     * Shizuku UserServices are separate uid=shell processes. Reinstalling or
     * crashing Unity does not necessarily unbind them, which left dozens of
     * DOZE VirtualDisplays and Moonlight tasks consuming RAM/compositor state.
     * A new primary service owns the new session, so it can safely reap older
     * MLOmega service processes before creating its first display. Slot services
     * do not run this path and therefore remain available for multi-window use.
     */
    private void reapAbandonedMLOmegaServices() {
        File proc = new File("/proc");
        File[] entries = proc.listFiles();
        if (entries == null) return;
        int self = Process.myPid();
        int killed = 0;
        for (File entry : entries) {
            if (entry == null || !entry.isDirectory()) continue;
            int pid;
            try { pid = Integer.parseInt(entry.getName()); }
            catch (NumberFormatException ignored) { continue; }
            if (pid == self) continue;
            String command = readProcCommandLine(pid);
            if (command == null || !command.startsWith("com.mlomega.xr.")) continue;
            if (!command.contains(":mlomega_trusted")
                    && !command.contains(":mlomega_app_slot_")) continue;
            try {
                Process.killProcess(pid);
                killed++;
                Log.i(TAG, "STALE_SERVICE_REAP pid=" + pid + " name=" + command);
            } catch (Throwable error) {
                Log.w(TAG, "stale service reap failed pid=" + pid, error);
            }
        }
        if (killed > 0) SystemClock.sleep(220L);
        Log.i(TAG, "STALE_SERVICE_REAP total=" + killed);
    }

    private static String readProcCommandLine(int pid) {
        File file = new File("/proc/" + pid + "/cmdline");
        try (FileInputStream stream = new FileInputStream(file)) {
            byte[] bytes = new byte[256];
            int count = stream.read(bytes);
            if (count <= 0) return null;
            int end = 0;
            while (end < count && bytes[end] != 0) end++;
            return new String(bytes, 0, end, "UTF-8");
        } catch (Throwable ignored) {
            return null;
        }
    }

    private String performRuntimePreflight() {
        // Only the explicit boot preflight may reap siblings. The same base
        // service is also used later by protected/cinema windows; doing this in
        // the constructor would kill legitimate multi-app slots mid-session.
        reapAbandonedMLOmegaServices();
        int dexSystem = putSetting("system", "dex_on_external_display", "0");
        int dexGlobal = putSetting("global", "dex_on_external_display", "0");
        int dexSecure = putSetting("secure", "dex_on_external_display", "0");
        return "OK:uid=" + Process.myUid()
                + ":dex=" + dexSystem + "," + dexGlobal + "," + dexSecure;
    }

    private int putSetting(String namespace, String key, String value) {
        try {
            java.lang.Process process = new ProcessBuilder(
                    "/system/bin/settings", "put", namespace, key, value)
                    .redirectErrorStream(true)
                    .start();
            return process.waitFor();
        } catch (Throwable error) {
            Log.w(TAG, "settings preflight failed " + namespace + "/" + key, error);
            return -1;
        }
    }

    private boolean reattachDisplaySurface(Surface surface) {
        if (surface == null || !surface.isValid() || virtualDisplay == null) {
            Log.w(TAG, "reattach rejected surface=" + surface
                    + " display=" + virtualDisplay);
            return false;
        }
        long identity = Binder.clearCallingIdentity();
        try {
            Surface previous = displaySurface;
            virtualDisplay.setSurface(surface);
            displaySurface = surface;
            if (previous != null && previous != surface) {
                try { previous.release(); } catch (Throwable ignored) {}
            }
            Log.i(TAG, "trusted display surface reattached id="
                    + virtualDisplay.getDisplay().getDisplayId());
            return true;
        } catch (Throwable error) {
            Log.e(TAG, "trusted display surface reattach failed", error);
            return false;
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    private boolean resizeDisplay(int width, int height, int dpi) {
        if (virtualDisplay == null) return false;
        int safeWidth = Math.max(640, Math.min(1920, width));
        int safeHeight = Math.max(360, Math.min(1600, height));
        int safeDpi = Math.max(160, dpi);
        long identity = Binder.clearCallingIdentity();
        try {
            virtualDisplay.resize(safeWidth, safeHeight, safeDpi);
            Log.i(TAG, "trusted display resized id="
                    + virtualDisplay.getDisplay().getDisplayId() + " size="
                    + safeWidth + "x" + safeHeight + " dpi=" + safeDpi);
            return true;
        } catch (Throwable error) {
            Log.e(TAG, "trusted display resize failed", error);
            return false;
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    @SuppressWarnings({"rawtypes", "unchecked"})
    private String startSecureStereoTask(
            Surface spatialSurface,
            int surfaceWidth,
            int surfaceHeight,
            String packageName,
            String uri) throws Exception {
        if (spatialSurface == null || !spatialSurface.isValid()) {
            throw new IllegalArgumentException("XREAL spatial surface invalid");
        }
        if (packageName == null || packageName.trim().isEmpty()) {
            throw new IllegalArgumentException("secure target package required");
        }

        releaseSecureTaskOrganizer();
        secureSpatialSurface = spatialSurface;
        secureTargetPackage = packageName;
        securePhysicalDisplayId = -1;
        // Keep the legitimate Android task and its input channel on display 0.
        // Only a protected mirror is parented under XREAL's own protected quad;
        // XREAL then performs the stereo fusion and world-space pose itself.
        secureTaskDisplayId = 0;
        secureTaskWidth = 1080;
        secureTaskHeight = 607;
        secureEyeWidth = Math.max(1, surfaceWidth);
        secureEyeHeight = Math.max(1, surfaceHeight);

        long identity = Binder.clearCallingIdentity();
        try {
            secureTaskOrganizer = new TaskOrganizer() {
                @Override
                public void onTaskAppeared(
                        ActivityManager.RunningTaskInfo taskInfo,
                        SurfaceControl leash) {
                    handleSecureTaskAppeared(taskInfo, leash);
                }

                @Override
                public void onTaskInfoChanged(ActivityManager.RunningTaskInfo taskInfo) {
                    if (!taskMatchesPackage(taskInfo)) return;
                    if (secureTaskLeash == null
                            && taskInfo != null
                            && taskInfo.taskId == secureRootTaskId
                            && secureRootLeash != null
                            && secureRootLeash.isValid()) {
                        handleSecureTaskAppeared(taskInfo, secureRootLeash);
                    } else if (secureLeftMirror != null) {
                        applySecureTaskTransform(
                                taskInfo, secureLeftMirror, secureRightMirror);
                    }
                }

                @Override
                public void onTaskVanished(ActivityManager.RunningTaskInfo taskInfo) {
                    if (taskMatchesPackage(taskInfo)) releaseSecureTaskSurfaces();
                }
            };
            List existing = secureTaskOrganizer.registerOrganizer();
            Log.i(TAG, "secure TaskOrganizer registered existing="
                    + (existing == null ? -1 : existing.size())
                    + " uid=" + Process.myUid());

            secureRootCookie = new Binder();
            secureRootReady = new CountDownLatch(1);
            createSecureRootTask();
            if (!secureRootReady.await(3500L, TimeUnit.MILLISECONDS)
                    || secureRootTaskId < 0 || secureRootToken == null) {
                throw new IllegalStateException(
                        "secure child root unavailable display="
                                + secureTaskDisplayId);
            }
            resizeSecureRoot();
            String output = launchIntoSecureRoot(packageName, uri);
            return "child_root=" + secureRootTaskId
                    + ":display=" + securePhysicalDisplayId
                    + ":task_display=" + secureTaskDisplayId
                    + ":launch_requested:" + packageName
                    + ":" + output;
        } catch (Throwable error) {
            releaseSecureTaskOrganizer();
            throw error;
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    private void invokeActivityOption(
            ActivityOptions options,
            String method,
            Class<?>[] parameterTypes,
            Object[] arguments) {
        try {
            Method target = ActivityOptions.class.getMethod(method, parameterTypes);
            target.invoke(options, arguments);
        } catch (Throwable error) {
            Log.w(TAG, "ActivityOptions." + method + " unavailable", error);
        }
    }

    private boolean taskMatchesPackage(ActivityManager.RunningTaskInfo info) {
        if (info == null || secureTargetPackage == null) return false;
        ComponentName top = info.topActivity;
        ComponentName base = info.baseActivity;
        return (top != null && secureTargetPackage.equals(top.getPackageName()))
                || (base != null && secureTargetPackage.equals(base.getPackageName()));
    }

    private void handleSecureTaskAppeared(
            ActivityManager.RunningTaskInfo taskInfo,
            SurfaceControl leash) {
        String component = taskInfo == null || taskInfo.topActivity == null
                ? "null" : taskInfo.topActivity.flattenToShortString();
        Log.i(TAG, "TaskOrganizer appeared task="
                + (taskInfo == null ? -1 : taskInfo.taskId)
                + " component=" + component
                + " leash=" + leash);
        if (taskContainsLaunchCookie(taskInfo, secureRootCookie)) {
            secureRootTaskId = taskInfo.taskId;
            secureRootToken = readTaskToken(taskInfo);
            secureRootLeash = leash;
            Log.i(TAG, "secure child root ready task=" + secureRootTaskId
                    + " display=" + securePhysicalDisplayId
                    + " token=" + secureRootToken);
            CountDownLatch ready = secureRootReady;
            if (ready != null) ready.countDown();
            return;
        }
        if (!taskMatchesPackage(taskInfo) || leash == null || !leash.isValid()) return;
        try {
            secureTaskLeash = leash;
            Method mirrorMethod = SurfaceControl.class.getDeclaredMethod(
                    "mirrorSurface", SurfaceControl.class);
            mirrorMethod.setAccessible(true);
            secureLeftMirror = (SurfaceControl) mirrorMethod.invoke(null, leash);
            secureRightMirror = null;
            if (secureLeftMirror == null || !secureLeftMirror.isValid()) {
                throw new IllegalStateException("secure task mirror invalid");
            }
            // Never move the real task leash away from display 0. Android 16
            // otherwise migrates its input handle to the physical XREAL display
            // while WindowManager still owns the task on display 0, leaving no
            // touchable window on either display. Two protected mirrors preserve
            // the secure composition and keep the original task/input channel
            // attached to its logical display. XREAL duplicates this one quad
            // correctly for both eyes, unlike two Android Presentations.
            applySecureTaskTransform(taskInfo, secureLeftMirror, secureRightMirror);
            Log.i(TAG, "secure spatial task attached task=" + taskInfo.taskId
                    + " package=" + secureTargetPackage);
        } catch (Throwable error) {
            Log.e(TAG, "secure stereo task attach failed", error);
            releaseSecureTaskSurfaces();
        }
    }

    private boolean taskContainsLaunchCookie(
            ActivityManager.RunningTaskInfo info,
            IBinder cookie) {
        if (info == null || cookie == null) return false;
        try {
            Method method = Class.forName("android.app.TaskInfo")
                    .getDeclaredMethod("containsLaunchCookie", IBinder.class);
            method.setAccessible(true);
            return Boolean.TRUE.equals(method.invoke(info, cookie));
        } catch (Throwable methodError) {
            try {
                Field field = Class.forName("android.app.TaskInfo")
                        .getDeclaredField("launchCookies");
                field.setAccessible(true);
                Object value = field.get(info);
                return value instanceof List && ((List<?>) value).contains(cookie);
            } catch (Throwable fieldError) {
                Log.w(TAG, "root launch-cookie inspection failed", fieldError);
                return false;
            }
        }
    }

    private Object readTaskToken(ActivityManager.RunningTaskInfo info) {
        if (info == null) return null;
        try {
            Field token = Class.forName("android.app.TaskInfo").getField("token");
            return token.get(info);
        } catch (Throwable publicError) {
            try {
                Field token = Class.forName("android.app.TaskInfo")
                        .getDeclaredField("token");
                token.setAccessible(true);
                return token.get(info);
            } catch (Throwable error) {
                Log.e(TAG, "root task token unavailable", error);
                return null;
            }
        }
    }

    private void createSecureRootTask() throws Exception {
        try {
            Method modern = TaskOrganizer.class.getDeclaredMethod(
                    "createRootTask",
                    int.class,
                    int.class,
                    IBinder.class,
                    boolean.class);
            modern.setAccessible(true);
            modern.invoke(
                    secureTaskOrganizer,
                    secureTaskDisplayId,
                    6,
                    secureRootCookie,
                    true);
            Log.i(TAG, "createRootTask signature=4");
            return;
        } catch (NoSuchMethodException unavailable) {
            Log.i(TAG, "createRootTask signature=4 unavailable; trying 3");
        }
        Method legacy = TaskOrganizer.class.getDeclaredMethod(
                "createRootTask",
                int.class,
                int.class,
                IBinder.class);
        legacy.setAccessible(true);
        legacy.invoke(
                secureTaskOrganizer,
                secureTaskDisplayId,
                6,
                secureRootCookie);
        Log.i(TAG, "createRootTask signature=3");
    }

    private String launchIntoSecureRoot(
            String packageName,
            String uri) throws Exception {
        Context context = obtainShellContext();
        if (context == null) {
            throw new IllegalStateException("shell context unavailable");
        }
        Intent intent = new Intent(Intent.ACTION_VIEW, Uri.parse(uri));
        intent.setPackage(packageName);
        intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK
                | Intent.FLAG_ACTIVITY_MULTIPLE_TASK);

        ActivityOptions options = ActivityOptions.makeBasic();
        options.setLaunchDisplayId(secureTaskDisplayId);
        options.setLaunchBounds(new Rect(
                0, 0, secureTaskWidth, secureTaskHeight));
        invokeRequiredActivityOption(
                options,
                "setLaunchWindowingMode",
                new Class<?>[] { int.class },
                new Object[] { 6 });
        try {
            Class<?> tokenClass = Class.forName(
                    "android.window.WindowContainerToken");
            invokeRequiredActivityOption(
                    options,
                    "setLaunchRootTask",
                    new Class<?>[] { tokenClass },
                    new Object[] { secureRootToken });
            Log.i(TAG, "secure launch target=root_token task=" + secureRootTaskId);
        } catch (NoSuchMethodException unavailable) {
            invokeRequiredActivityOption(
                    options,
                    "setLaunchTaskId",
                    new Class<?>[] { int.class },
                    new Object[] { secureRootTaskId });
            Log.i(TAG, "secure launch target=task_id task=" + secureRootTaskId);
        }
        try {
            invokeRequiredActivityOption(
                    options,
                    "setRemoveWithTaskOrganizer",
                    new Class<?>[] { boolean.class },
                    new Object[] { true });
        } catch (NoSuchMethodException unavailable) {
            Log.i(TAG, "setRemoveWithTaskOrganizer unavailable");
        }

        PendingIntent pending = PendingIntent.getActivity(
                context,
                (int) (SystemClock.uptimeMillis() & 0x7fffffff),
                intent,
                PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_MUTABLE,
                options.toBundle());
        pending.send(
                context,
                0,
                null,
                null,
                null,
                null,
                options.toBundle());
        Log.i(TAG, "secure PendingIntent sent creator="
                + pending.getCreatorPackage()
                + " uid=" + pending.getCreatorUid()
                + " taskDisplay=" + secureTaskDisplayId
                + " physicalDisplay=" + securePhysicalDisplayId);
        return "pending_intent_sent:" + pending.getCreatorPackage();
    }

    private void invokeRequiredActivityOption(
            ActivityOptions options,
            String method,
            Class<?>[] parameterTypes,
            Object[] arguments) throws Exception {
        Method target = ActivityOptions.class.getDeclaredMethod(method, parameterTypes);
        target.setAccessible(true);
        target.invoke(options, arguments);
    }

    private void applySecureTaskTransform(
            ActivityManager.RunningTaskInfo taskInfo,
            SurfaceControl leftLeash,
            SurfaceControl rightMirror) {
        if (leftLeash == null || !leftLeash.isValid()
                || secureSpatialSurface == null || !secureSpatialSurface.isValid()) return;
        Rect taskBounds = readTaskBounds(taskInfo);
        int sourceWidth = Math.max(1, taskBounds.width());
        int sourceHeight = Math.max(1, taskBounds.height());
        String result = SecureTaskSurfaceProbe.attachMirror(
                secureSpatialSurface,
                leftLeash,
                sourceWidth,
                sourceHeight,
                secureEyeWidth,
                secureEyeHeight);
        if (result == null || !result.startsWith("spatial_mirror_attached:")) {
            throw new IllegalStateException("spatial mirror rejected: " + result);
        }
        Log.i(TAG, result);
    }

    private Rect readTaskBounds(ActivityManager.RunningTaskInfo taskInfo) {
        try {
            Field configurationField = Class.forName("android.app.TaskInfo")
                    .getDeclaredField("configuration");
            configurationField.setAccessible(true);
            Object configuration = configurationField.get(taskInfo);
            Field windowConfigurationField = configuration.getClass()
                    .getDeclaredField("windowConfiguration");
            windowConfigurationField.setAccessible(true);
            Object windowConfiguration = windowConfigurationField.get(configuration);
            Method getBounds = windowConfiguration.getClass().getMethod("getBounds");
            Rect bounds = (Rect) getBounds.invoke(windowConfiguration);
            if (bounds != null && !bounds.isEmpty()) return new Rect(bounds);
        } catch (Throwable error) {
            Log.w(TAG, "task bounds reflection unavailable; using S24 bounds", error);
        }
        return new Rect(0, 0, 1080, 2340);
    }

    private void releaseSecureTaskSurfaces() {
        try {
            SurfaceControl.Transaction transaction = new SurfaceControl.Transaction();
            if (secureLeftMirror != null && secureLeftMirror.isValid()) {
                transaction.reparent(secureLeftMirror, null);
            }
            if (secureRightMirror != null && secureRightMirror.isValid()) {
                transaction.reparent(secureRightMirror, null);
            }
            transaction.apply();
        } catch (Throwable error) {
            Log.w(TAG, "secure task surface release failed", error);
        }
        try { SecureTaskSurfaceProbe.releaseMirrorParent(); }
        catch (Throwable error) {
            Log.w(TAG, "secure spatial parent release failed", error);
        }
        if (secureLeftMirror != null) {
            try { secureLeftMirror.release(); } catch (Throwable ignored) {}
        }
        if (secureRightMirror != null) {
            try { secureRightMirror.release(); } catch (Throwable ignored) {}
        }
        secureTaskLeash = null;
        secureLeftMirror = null;
        secureRightMirror = null;
    }

    /**
     * Overlay-free DRM handoff. The physical XREAL panel is retargeted to the
     * trusted VirtualDisplay layer stack, so protected buffers stay entirely in
     * SurfaceFlinger. Navigation remains spatial, commercial playback becomes a
     * temporary system-owned cinema screen, and stopSystemMirror restores the
     * exact native panel layer stack captured here.
     */
    private String maybeStartSystemMirror(
            int glassesDisplayId,
            int virtualDisplayId,
            String packageName,
            boolean force) throws Exception {
        if (glassesDisplayId <= 0) {
            return "WAIT:no_external_display:" + glassesDisplayId;
        }
        if (virtualDisplayId < 0 || virtualDisplay == null) {
            return "WAIT:no_virtual_display:" + virtualDisplayId;
        }
        if (!force && !isProtectedPlaybackActive(packageName)) {
            return "WAIT:not_playing:" + packageName;
        }
        if (mirroredGlassesDisplayId == glassesDisplayId
                && mirroredVirtualDisplayId == virtualDisplayId
                && originalPanelLayerStack >= 0) {
            return "ALREADY:glasses=" + glassesDisplayId
                    + ":vd=" + virtualDisplayId;
        }

        long identity = Binder.clearCallingIdentity();
        try {
            Class<?> surfaceControl = Class.forName("android.view.SurfaceControl");
            IBinder panelToken = resolveExternalPanelToken(surfaceControl);
            if (panelToken == null) return "WAIT:no_external_panel_token";

            Integer virtualLayerStack = readLayerStack(virtualDisplayId);
            Integer panelLayerStack = readLayerStack(glassesDisplayId);
            if (virtualLayerStack == null) {
                return "WAIT:no_vd_layer_stack:" + virtualDisplayId;
            }
            if (panelLayerStack == null) {
                return "WAIT:no_panel_layer_stack:" + glassesDisplayId;
            }
            if (originalPanelLayerStack < 0) {
                originalPanelLayerStack = panelLayerStack;
            }
            setPanelLayerStack(panelToken, virtualLayerStack);

            mirroredGlassesDisplayId = glassesDisplayId;
            mirroredVirtualDisplayId = virtualDisplayId;
            cinemaProjectionPreset = 0;
            String projection = applyCinemaProjectionLocked(
                    panelToken, glassesDisplayId, virtualDisplayId, 0);
            String result = "STARTED:glasses=" + glassesDisplayId
                    + ":panelLS=" + originalPanelLayerStack
                    + ":vd=" + virtualDisplayId
                    + ":vdLS=" + virtualLayerStack
                    + ":projection=" + projection
                    + ":package=" + packageName;
            Log.w(TAG, "SYSTEM_CINEMA " + result);
            return result;
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    private String stopSystemMirror() {
        int glassesDisplayId = mirroredGlassesDisplayId;
        int restoreLayerStack = originalPanelLayerStack;
        if (glassesDisplayId < 0 || restoreLayerStack < 0) {
            return "SKIP:not_mirroring";
        }
        long identity = Binder.clearCallingIdentity();
        try {
            String navigation = returnMirroredAppToBrowse();
            Class<?> surfaceControl = Class.forName("android.view.SurfaceControl");
            IBinder panelToken = resolveExternalPanelToken(surfaceControl);
            if (panelToken == null) return "FAIL:no_external_panel_token";
            applyCinemaProjectionLocked(
                    panelToken, glassesDisplayId, mirroredVirtualDisplayId, 0);
            setPanelLayerStack(panelToken, restoreLayerStack);
            String result = "RESTORED:glasses=" + glassesDisplayId
                    + ":layerStack=" + restoreLayerStack
                    + ":navigation=" + navigation;
            Log.w(TAG, "SYSTEM_CINEMA " + result);
            mirroredGlassesDisplayId = -1;
            mirroredVirtualDisplayId = -1;
            originalPanelLayerStack = -1;
            return result;
        } catch (Throwable error) {
            Log.e(TAG, "system cinema restore failed", error);
            return "FAIL:" + error.getClass().getSimpleName()
                    + ":" + String.valueOf(error.getMessage());
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    private String returnMirroredAppToBrowse() {
        int displayId = mirroredVirtualDisplayId;
        if (displayId < 0) return "SKIP:no_virtual_display";
        try {
            String result = runCommand(
                    "/system/bin/input",
                    "-d", String.valueOf(displayId),
                    "keyevent", "4").trim();
            // Let Netflix leave PlayerActivity and draw its non-protected browse
            // screen before XREAL returns to the protected world-space quad.
            SystemClock.sleep(650L);
            Log.i(TAG, "cinema app returned to browse display=" + displayId
                    + " result=" + result);
            return "OK";
        } catch (Throwable error) {
            Log.w(TAG, "cinema browse return failed", error);
            return "FAIL:" + error.getClass().getSimpleName();
        }
    }

    private String toggleMirroredMedia() {

        }
    }

    private String seekMirroredMediaBack(long milliseconds) {
        long amount = Math.max(1000L, Math.min(60000L, milliseconds));
        try {
            Context context = obtainShellContext();
            MediaSessionManager manager = context == null ? null
                    : (MediaSessionManager) context.getSystemService(
                            Context.MEDIA_SESSION_SERVICE);
            List<MediaController> controllers = manager == null
                    ? null : manager.getActiveSessions(null);
            MediaController chosen = null;
            if (controllers != null) {
                for (MediaController controller : controllers) {
                    if (controller == null) continue;
                    PlaybackState candidate = controller.getPlaybackState();
                    if (secureTargetPackage != null
                            && secureTargetPackage.equals(controller.getPackageName())) {
                        chosen = controller;
                        break;
                    }
                    if (chosen == null && candidate != null
                            && (candidate.getState() == PlaybackState.STATE_PLAYING
                            || candidate.getState() == PlaybackState.STATE_BUFFERING)) {
                        chosen = controller;
                    }
                }
            }
            PlaybackState state = chosen == null ? null : chosen.getPlaybackState();
            if (chosen != null && state != null && state.getPosition() >= 0L) {
                long position = state.getPosition();
                if (state.getState() == PlaybackState.STATE_PLAYING
                        && state.getLastPositionUpdateTime() > 0L) {
                    long elapsed = Math.max(0L,
                            SystemClock.elapsedRealtime()
                                    - state.getLastPositionUpdateTime());
                    position += (long) (elapsed * state.getPlaybackSpeed());
                }
                long target = Math.max(0L, position - amount);
                chosen.getTransportControls().seekTo(target);
                Log.i(TAG, "cinema media seek " + position + " -> " + target
                        + " package=" + chosen.getPackageName());
                return "OK:" + target;
            }
        } catch (Throwable error) {
            Log.w(TAG, "cinema exact seek failed; trying media rewind", error);
        }
        try {
            // Provider fallback: not every commercial app publishes seekTo to
            // uid=shell, but all Android players understand MEDIA_REWIND.
            runCommand("/system/bin/input", "keyevent", "89");
            return "FALLBACK:MEDIA_REWIND";
        } catch (Throwable error) {
            Log.w(TAG, "cinema media rewind failed", error);
            return "FAIL:" + error.getClass().getSimpleName();
        }
    }

    private IBinder resolveExternalPanelToken(Class<?> surfaceControl) {
        try {
            long[] physicalIds = (long[]) surfaceControl
                    .getMethod("getPhysicalDisplayIds")
                    .invoke(null);
            if (physicalIds == null || physicalIds.length < 2) return null;
            long internalId = physicalIds[0];
            Method getToken = surfaceControl.getMethod(
                    "getPhysicalDisplayToken", long.class);
            for (int index = physicalIds.length - 1; index >= 0; index--) {
                long physicalId = physicalIds[index];
                if (physicalId == internalId) continue;
                IBinder token = (IBinder) getToken.invoke(null, physicalId);
                if (token != null) return token;
            }
        } catch (Throwable error) {
            Log.e(TAG, "external panel token unavailable", error);
        }
        return null;
    }

    private Integer readLayerStack(int logicalDisplayId) {
        try {
            Class<?> globalClass = Class.forName(
                    "android.hardware.display.DisplayManagerGlobal");
            Object global = globalClass.getMethod("getInstance").invoke(null);
            Object info = global.getClass()
                    .getMethod("getDisplayInfo", int.class)
                    .invoke(global, logicalDisplayId);
            if (info == null) return null;
            return info.getClass().getField("layerStack").getInt(info);
        } catch (Throwable error) {
            Log.e(TAG, "layerStack unavailable display=" + logicalDisplayId, error);
            return null;
        }
    }

    private void setPanelLayerStack(IBinder panelToken, int layerStack)
            throws Exception {
        Class<?> transactionClass = Class.forName(
                "android.view.SurfaceControl$Transaction");
        Object transaction = transactionClass.getConstructor().newInstance();
        try {
            transactionClass.getMethod(
                    "setDisplayLayerStack", IBinder.class, int.class)
                    .invoke(transaction, panelToken, layerStack);
            transactionClass.getMethod("apply").invoke(transaction);
        } finally {
            try { transactionClass.getMethod("close").invoke(transaction); }
            catch (Throwable ignored) {}
        }
    }

    private String applyCinemaProjection(int preset) {
        int glassesDisplayId = mirroredGlassesDisplayId;
        int virtualDisplayId = mirroredVirtualDisplayId;
        if (glassesDisplayId < 0 || virtualDisplayId < 0) {
            return "SKIP:not_mirroring";
        }
        long identity = Binder.clearCallingIdentity();
        try {
            Class<?> surfaceControl = Class.forName("android.view.SurfaceControl");
            IBinder panelToken = resolveExternalPanelToken(surfaceControl);
            if (panelToken == null) return "FAIL:no_external_panel_token";
            cinemaProjectionPreset = Math.max(0, Math.min(2, preset));
            return applyCinemaProjectionLocked(
                    panelToken,
                    glassesDisplayId,
                    virtualDisplayId,
                    cinemaProjectionPreset);
        } catch (Throwable error) {
            Log.e(TAG, "cinema projection failed", error);
            return "FAIL:" + error.getClass().getSimpleName();
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    private String applyCinemaProjectionLocked(
            IBinder panelToken,
            int glassesDisplayId,
            int virtualDisplayId,
            int preset) throws Exception {
        int[] source = readDisplaySize(virtualDisplayId);
        int[] panel = readDisplaySize(glassesDisplayId);
        int sourceWidth = source[0] > 0 ? source[0] : 1920;
        int sourceHeight = source[1] > 0 ? source[1] : 1080;
        int panelWidth = panel[0] > 0 ? panel[0] : 1920;
        int panelHeight = panel[1] > 0 ? panel[1] : 1080;

        float fit = Math.min(
                panelWidth / (float) sourceWidth,
                panelHeight / (float) sourceHeight);
        float scale = preset == 2 ? 0.70f : (preset == 1 ? 0.85f : 1.0f);
        int width = Math.max(1, Math.round(sourceWidth * fit * scale));
        int height = Math.max(1, Math.round(sourceHeight * fit * scale));
        int left = (panelWidth - width) / 2;
        int top = (panelHeight - height) / 2;
        Rect src = new Rect(0, 0, sourceWidth, sourceHeight);
        Rect dst = new Rect(left, top, left + width, top + height);

        Class<?> transactionClass = Class.forName(
                "android.view.SurfaceControl$Transaction");
        Object transaction = transactionClass.getConstructor().newInstance();
        try {
            transactionClass.getMethod(
                    "setDisplayProjection",
                    IBinder.class,
                    int.class,
                    Rect.class,
                    Rect.class)
                    .invoke(transaction, panelToken, 0, src, dst);
            transactionClass.getMethod("apply").invoke(transaction);
        } finally {
            try { transactionClass.getMethod("close").invoke(transaction); }
            catch (Throwable ignored) {}
        }
        String result = "OK:preset=" + preset + ":src=" + src + ":dst=" + dst;
        Log.w(TAG, "CINEMA_PROJECTION " + result);
        return result;
    }

    private int[] readDisplaySize(int displayId) {
        try {
            Context context = obtainShellContext();
            DisplayManager manager = context == null ? null
                    : (DisplayManager) context.getSystemService(Context.DISPLAY_SERVICE);
            android.view.Display display = manager == null
                    ? null : manager.getDisplay(displayId);
            if (display != null) {
                // XREAL One can expose a 640x480 hardware mode while Android's
                // actual application surface is 1920x1080. PortalPad's proven
                // DRM-safe projection path intentionally uses getRealMetrics.
                android.util.DisplayMetrics metrics =
                        new android.util.DisplayMetrics();
                display.getRealMetrics(metrics);
                if (metrics.widthPixels > 0 && metrics.heightPixels > 0) {
                    return new int[] {
                            metrics.widthPixels, metrics.heightPixels};
                }
            }
            android.view.Display.Mode mode = display == null ? null : display.getMode();
            if (mode != null) {
                return new int[] {
                        mode.getPhysicalWidth(), mode.getPhysicalHeight()};
            }
        } catch (Throwable error) {
            Log.w(TAG, "display size unavailable id=" + displayId, error);
        }
        return new int[] {-1, -1};
    }

    private boolean isProtectedPlaybackActive(String packageName) {
        if (packageName == null || packageName.trim().isEmpty()) return false;
        String needle = packageName.toLowerCase();
        try {
            String sessions = runCommand("/system/bin/dumpsys", "media_session")
                    .toLowerCase();
            int offset = 0;
            while ((offset = sessions.indexOf(needle, offset)) >= 0) {
                int start = Math.max(0, offset - 1800);
                int end = Math.min(sessions.length(), offset + 2600);
                String block = sessions.substring(start, end);
                if (block.contains("state=3") || block.contains("state=6")
                        || block.contains("state=buffering")
                        || block.contains("state=playing")) {
                    return true;
                }
                offset += needle.length();
            }

            // Netflix may publish its media session a few frames after the
            // player Activity appears. This fallback performs the handoff as
            // soon as the real player screen is resumed, never on browse/detail.
            String activities = runCommand(
                    "/system/bin/dumpsys", "activity", "activities")
                    .toLowerCase();
            for (String line : activities.split("\\n")) {
                if (!line.contains(needle)) continue;
                if ((line.contains("resumed") || line.contains("topactivity"))
                        && (line.contains("player")
                        || line.contains("playback")
                        || line.contains("watch"))) {
                    return true;
                }
            }
        } catch (Throwable error) {
            Log.w(TAG, "playback probe failed package=" + packageName, error);
        }
        return false;
    }

    private String runCommand(String... command) throws Exception {
        ProcessBuilder builder = new ProcessBuilder(command);
        builder.redirectErrorStream(true);
        java.lang.Process process = builder.start();
        StringBuilder output = new StringBuilder();
        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(process.getInputStream()))) {
            String line;
            while ((line = reader.readLine()) != null) {
                output.append(line).append('\n');
            }
        }
        process.waitFor();
        return output.toString();
    }

    private String launchPackagedUri(
            int displayId,
            String packageName,
            String uri) throws Exception {
        if (packageName == null || packageName.trim().isEmpty()) {
            throw new IllegalArgumentException("package name required");
        }
        long identity = Binder.clearCallingIdentity();
        try {
            // A singleTask/singleInstance activity already alive on display 0
            // may ignore --display and simply reuse its phone task. Start from
            // a stopped package so Android must create a fresh task on the
            // trusted VirtualDisplay. Login/data are preserved by force-stop.
            forceStopPackage(packageName);
            SystemClock.sleep(180L);
            String deepLinkResult = null;
            if (uri != null && !uri.trim().isEmpty()) {
                deepLinkResult = runActivityManagerStart(
                        "/system/bin/am",
                        "start",
                        "--display", String.valueOf(displayId),
                        "-f", "0x18000000",
                        "-a", "android.intent.action.VIEW",
                        "-d", uri,
                        "-p", packageName);
                if (!activityLaunchFailed(deepLinkResult)) {
                    Log.i(TAG, "APP_LAUNCH_RECEIPT package=" + packageName
                            + " display=" + displayId
                            + " route=deep_link result=" + deepLinkResult);
                    return "route=deep_link " + deepLinkResult;
                }
                Log.w(TAG, "deep-link unsupported; falling back to launcher package="
                        + packageName + " result=" + deepLinkResult);
            }

            // Not every Android application declares an ACTION_VIEW handler for
            // its own web URL. The Google app on the validated S24 is one such
            // case: its real entry point is SearchActivity. MAIN/LAUNCHER keeps
            // the transport package-agnostic and also covers Spotify and future
            // applications without hard-coding their activity class names.
            // These two applications expose several MAIN activities on the
            // validated S24. Android 16 rejects the package-only launcher
            // intent from a shell UserService, although the APK is installed.
            // Pin only their official launcher components; every other app
            // keeps the previously validated generic route unchanged.
            String explicitLauncher = null;
            if ("com.spotify.music".equals(packageName)) {
                explicitLauncher = "com.spotify.music/.MainActivity";
            } else if ("com.amazon.avod.thirdpartyclient".equals(packageName)) {
                explicitLauncher =
                        "com.amazon.avod.thirdpartyclient/.LauncherActivity";
            } else if ("com.limelight".equals(packageName)) {
                explicitLauncher = "com.limelight/.PcView";
            }
            String launcherResult = explicitLauncher == null
                    ? runActivityManagerStart(
                            "/system/bin/am",
                            "start",
                            "--display", String.valueOf(displayId),
                            "-f", "0x18000000",
                            "-a", "android.intent.action.MAIN",
                            "-c", "android.intent.category.LAUNCHER",
                            "-p", packageName)
                    : runActivityManagerStart(
                            "/system/bin/am",
                            "start",
                            "--display", String.valueOf(displayId),
                            "-f", "0x18000000",
                            "-n", explicitLauncher);
            if (activityLaunchFailed(launcherResult)) {
                throw new IllegalStateException(
                        "application launch failed package=" + packageName
                                + " deepLink=" + deepLinkResult
                                + " launcher=" + launcherResult);
            }
            Log.i(TAG, "APP_LAUNCH_RECEIPT package=" + packageName
                    + " display=" + displayId
                    + " route=launcher result=" + launcherResult);
            return "route=launcher " + launcherResult;
        } finally {
            Binder.restoreCallingIdentity(identity);
        }
    }

    private String runActivityManagerStart(String... command) throws Exception {
        ProcessBuilder builder = new ProcessBuilder(command);
        builder.redirectErrorStream(true);
        java.lang.Process process = builder.start();
        StringBuilder output = new StringBuilder();
        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(process.getInputStream()))) {
            String line;
            while ((line = reader.readLine()) != null) {
                if (output.length() > 0) output.append(' ');
                output.append(line);
            }
        }
        int exit = process.waitFor();
        return "exit=" + exit + " " + output;
    }

    private boolean activityLaunchFailed(String result) {
        if (result == null || result.trim().isEmpty()) return true;
        String normalized = result.toLowerCase(java.util.Locale.ROOT);
        return !normalized.startsWith("exit=0")
                || normalized.contains("error:")
                || normalized.contains("unable to resolve")
                || normalized.contains("no activity found")
                || normalized.contains("does not exist")
                || normalized.contains("security exception");
    }

    private boolean forceStopPackage(String packageName) {
        if (packageName == null ||
                !packageName.matches("[A-Za-z0-9_]+(?:\\.[A-Za-z0-9_]+)+")) {
            Log.w(TAG, "refusing invalid force-stop package=" + packageName);
            return false;
        }
        try {
            ProcessBuilder builder = new ProcessBuilder(
                    "/system/bin/am", "force-stop", "--user", "0", packageName);
            builder.redirectErrorStream(true);
            java.lang.Process process = builder.start();
            int exit = process.waitFor();
            String remainingPids = "";
            for (int attempt = 0; attempt < 6; attempt++) {
                remainingPids = runCommand(
                        "/system/bin/pidof", packageName).trim();
                if (remainingPids.isEmpty()) break;
                SystemClock.sleep(60L);
            }
            boolean stopped = exit == 0 && remainingPids.isEmpty();
            Log.i(TAG, "APP_STOP_RECEIPT package=" + packageName
                    + " forceStopExit=" + exit
                    + " processAbsent=" + remainingPids.isEmpty()
                    + (remainingPids.isEmpty() ? "" : " pids=" + remainingPids));
            return stopped;
        } catch (Throwable error) {
            Log.e(TAG, "force-stop failed package=" + packageName, error);
            return false;
        }
    }

    private Context obtainShellContext() {
        try {
            Class<?> activityThread = Class.forName("android.app.ActivityThread");
            Object thread = activityThread.getMethod("currentActivityThread").invoke(null);
            if (thread != null) {
                Context system = (Context) activityThread
                        .getMethod("getSystemContext")
                        .invoke(thread);
                if (system != null) {
                    String[] packages = system.getPackageManager()
                            .getPackagesForUid(Process.myUid());
                    if (packages != null && packages.length > 0) {
                        return system.createPackageContext(
                                packages[0], Context.CONTEXT_IGNORE_SECURITY);
                    }
                    return system;
                }
            }
        } catch (Throwable error) {
            Log.w(TAG, "system context fallback", error);
        }
        return suppliedContext;
    }

    private void setupInputInjection() {
        try {
            try {
                Class<?> global = Class.forName(
                        "android.hardware.input.InputManagerGlobal");
                inputManager = global.getMethod("getInstance").invoke(null);
                injectInputEvent = global.getMethod(
                        "injectInputEvent", InputEvent.class, int.class);
            } catch (Throwable modernUnavailable) {
                Class<?> legacy = Class.forName(
                        "android.hardware.input.InputManager");
                inputManager = legacy.getMethod("getInstance").invoke(null);
                injectInputEvent = legacy.getMethod(
                        "injectInputEvent", InputEvent.class, int.class);
            }
            setMotionDisplayId = MotionEvent.class.getMethod(
                    "setDisplayId", int.class);
            inputReflectionBroken = inputManager == null ||
                    injectInputEvent == null || setMotionDisplayId == null;
            Log.i(TAG, "input injection ready=" + !inputReflectionBroken);
        } catch (Throwable error) {
            inputReflectionBroken = true;
            Log.e(TAG, "input injection reflection unavailable", error);
        }
    }

    private boolean injectPointer(
            int displayId,
            int action,
            float x,
            float y,
            long downTime) {
        if (inputReflectionBroken || displayId < 0) return false;
        long now = SystemClock.uptimeMillis();
        long gestureStart = downTime > 0L ? downTime : now;
        MotionEvent event = MotionEvent.obtain(
                gestureStart,
                now,
                action,
                x,
                y,
                0);
        try {
            event.setSource(InputDevice.SOURCE_TOUCHSCREEN);
            setMotionDisplayId.invoke(event, displayId);
            Object result = injectInputEvent.invoke(inputManager, event, 0);
            return !(result instanceof Boolean) || (Boolean) result;
        } catch (Throwable error) {
            Log.e(TAG, "pointer injection failed action=" + action +
                    " display=" + displayId + " x=" + x + " y=" + y,
                    error);
            return false;
        } finally {
            event.recycle();
        }
    }

    private boolean injectHover(int displayId, float x, float y) {
        if (inputReflectionBroken || displayId < 0) return false;
        long now = SystemClock.uptimeMillis();
        MotionEvent.PointerProperties properties =
                new MotionEvent.PointerProperties();
        properties.id = 0;
        properties.toolType = MotionEvent.TOOL_TYPE_MOUSE;
        MotionEvent.PointerCoords coordinates = new MotionEvent.PointerCoords();
        coordinates.x = x;
        coordinates.y = y;
        coordinates.pressure = 0f;
        coordinates.size = 1f;
        MotionEvent event = MotionEvent.obtain(
                now,
                now,
                MotionEvent.ACTION_HOVER_MOVE,
                1,
                new MotionEvent.PointerProperties[] { properties },
                new MotionEvent.PointerCoords[] { coordinates },
                0,
                0,
                1f,
                1f,
                0,
                0,
                InputDevice.SOURCE_MOUSE,
                0);
        try {
            setMotionDisplayId.invoke(event, displayId);
            Object result = injectInputEvent.invoke(inputManager, event, 0);
            return !(result instanceof Boolean) || (Boolean) result;
        } catch (Throwable error) {
            Log.e(TAG, "pointer hover injection failed display=" + displayId,
                    error);
            return false;
        } finally {
            event.recycle();
        }
    }

    private boolean openKeyboard(int displayId) {
        if (displayId < 0) return false;
        try {
            ProcessBuilder builder = new ProcessBuilder(
                    "/system/bin/input",
                    "-d", String.valueOf(displayId),
                    "keyevent", "KEYCODE_SEARCH");
            builder.redirectErrorStream(true);
            java.lang.Process process = builder.start();
            int exit = process.waitFor();
            Log.i(TAG, "keyboard/search display=" + displayId + " exit=" + exit);
            return exit == 0;
        } catch (Throwable error) {
            Log.e(TAG, "keyboard/search failed display=" + displayId, error);
            return false;
        }
    }

    private boolean inputText(int displayId, String text) {
        if (displayId < 0 || text == null || text.isEmpty()) return false;
        try {
            // Android's input command interprets %s as a literal space. Using
            // ProcessBuilder keeps the text out of a shell and avoids command
            // injection while still targeting the private VirtualDisplay.
            String encoded = text.replace("%", "%%").replace(" ", "%s");
            ProcessBuilder builder = new ProcessBuilder(
                    "/system/bin/input",
                    "-d", String.valueOf(displayId),
                    "text", encoded);
            builder.redirectErrorStream(true);
            java.lang.Process process = builder.start();
            int exit = process.waitFor();
            Log.i(TAG, "input text display=" + displayId + " exit=" + exit
                    + " chars=" + text.length());
            return exit == 0;
        } catch (Throwable error) {
            Log.e(TAG, "input text failed display=" + displayId, error);
            return false;
        }
    }

    private boolean inputKey(int displayId, int keyCode) {
        if (displayId < 0 || keyCode <= 0) return false;
        try {
            ProcessBuilder builder = new ProcessBuilder(
                    "/system/bin/input",
                    "-d", String.valueOf(displayId),
                    "keyevent", String.valueOf(keyCode));
            builder.redirectErrorStream(true);
            java.lang.Process process = builder.start();
            int exit = process.waitFor();
            Log.i(TAG, "input key display=" + displayId + " key=" + keyCode
                    + " exit=" + exit);
            return exit == 0;
        } catch (Throwable error) {
            Log.e(TAG, "input key failed display=" + displayId, error);
            return false;
        }
    }

    private boolean clearText(int displayId) {
        if (displayId < 0) return false;
        try {
            java.lang.Process selectAll = new ProcessBuilder(
                    "/system/bin/input",
                    "-d", String.valueOf(displayId),
                    "keycombination", "KEYCODE_CTRL_LEFT", "KEYCODE_A")
                    .redirectErrorStream(true)
                    .start();
            if (selectAll.waitFor() != 0) return false;
            java.lang.Process delete = new ProcessBuilder(
                    "/system/bin/input",
                    "-d", String.valueOf(displayId),
                    "keyevent", "KEYCODE_DEL")
                    .redirectErrorStream(true)
                    .start();
            return delete.waitFor() == 0;
        } catch (Throwable error) {
            Log.e(TAG, "clear text failed display=" + displayId, error);
            return false;
        }
    }

    private void terminateAfterReply() {
        final int pid = Process.myPid();
        new Thread(() -> {
            SystemClock.sleep(180L);
            Log.i(TAG, "USER_SERVICE_EXIT_RECEIPT pid=" + pid);
            Process.killProcess(pid);
        }, "mlomega-user-service-exit").start();
    }

    private void releaseDisplay() {
        stopSystemMirror();
        releaseSecureTaskOrganizer();
        if (virtualDisplay != null) {
            try { virtualDisplay.setSurface(null); } catch (Throwable ignored) {}
            try { virtualDisplay.release(); } catch (Throwable ignored) {}
            virtualDisplay = null;
        }
        if (displaySurface != null) {
            try { displaySurface.release(); } catch (Throwable ignored) {}
            displaySurface = null;
        }
    }

    @SuppressWarnings("unused")
    public void destroy() {
        releaseDisplay();
        System.exit(0);
    }
